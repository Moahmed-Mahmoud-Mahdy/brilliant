    10→// ─── Same hashing as src/lib/auth.ts ───
    11→function hashPassword(password: string): string {
    12→  const salt = randomBytes(16).toString("hex");
    13→  const hash = scryptSync(password, salt, 64).toString("hex");
    14→  return `${salt}:${hash}`;
    15→}

    18→async function receiveStock(skuId: string, qty: number, reason: string) {
    19→  const sku = await db.sku.findUniqueOrThrow({ where: { id: skuId } });
    20→  const before = sku.availableQty;
    21→  await db.sku.update({
    22→    where: { id: skuId },
    23→    data: { availableQty: before + qty },
    24→  });
    25→  await db.inventoryTransaction.create({
    26→    data: {
    27→      skuId,
    28→      type: "RECEIVE",
    29→      quantity: qty,
    30→      balanceBefore: before,
    31→      balanceAfter: before + qty,
    32→      reason,
    33→    },
    34→  });
    35→}
    36→
    37→async function reserveStock(skuId: string, qty: number, orderNumber: string, adminName: string) {
    38→  const sku = await db.sku.findUniqueOrThrow({ where: { id: skuId } });
    39→  const before = sku.availableQty;
    40→  await db.sku.update({
    41→    where: { id: skuId },
    42→    data: { availableQty: before - qty, reservedQty: sku.reservedQty + qty },
    43→  });
    44→  await db.inventoryTransaction.create({
    45→    data: {
    46→      skuId,
    47→      type: "RESERVE",
    48→      quantity: -qty,
    49→      balanceBefore: before,
    50→      balanceAfter: before - qty,
    51→      reason: `حجز كمية للطلب ${orderNumber}`,
    52→      orderId: orderNumber,
    53→      adminName,
    54→    },
    55→  });
    56→}
    57→
    58→async function releaseStock(skuId: string, qty: number, orderNumber: string, adminName: string) {
    59→  const sku = await db.sku.findUniqueOrThrow({ where: { id: skuId } });
    60→  const before = sku.availableQty;
    61→  await db.sku.update({
    62→    where: { id: skuId },
    63→    data: { availableQty: before + qty, reservedQty: Math.max(0, sku.reservedQty - qty) },
    64→  });
    65→  await db.inventoryTransaction.create({
    66→    data: {
    67→      skuId,
    68→      type: "RELEASE",
    69→      quantity: qty,
    70→      balanceBefore: before,
    71→      balanceAfter: before + qty,
    72→      reason: `تحرير حجز عند إلغاء الطلب ${orderNumber}`,
    73→      orderId: orderNumber,
    74→      adminName,
    75→    },
    76→  });
    77→}
    78→
    79→async function saleStock(skuId: string, qty: number, orderNumber: string, adminName: string) {
    80→  const sku = await db.sku.findUniqueOrThrow({ where: { id: skuId } });
    81→  await db.sku.update({
    82→    where: { id: skuId },
    83→    data: { reservedQty: Math.max(0, sku.reservedQty - qty), soldQty: sku.soldQty + qty },
    84→  });
    85→  await db.inventoryTransaction.create({
    86→    data: {
    87→      skuId,
    88→      type: "SALE",
    89→      quantity: -qty,
    90→      balanceBefore: sku.availableQty,
    91→      balanceAfter: sku.availableQty,
    92→      reason: `تثبيت بيع الطلب ${orderNumber} عند التسليم`,
    93→      orderId: orderNumber,
    94→      adminName,
    95→    },
    96→  });
    97→}

    99→function daysAgo(n: number, hours = 0): Date {
   100→  const d = new Date();
   101→  d.setDate(d.getDate() - n);
   102→  d.setHours(d.getHours() - hours);
   103→  return d;
   104→}

   613→    const d = daysAgo(daysBack);
   614→    const y = String(d.getFullYear()).slice(2);
   615→    const m = String(d.getMonth() + 1).padStart(2, "0");
   616→    const day = String(d.getDate()).padStart(2, "0");
   617→    return `BR-${y}${m}${day}-${String(++orderSeq).padStart(4, "0")}`;
